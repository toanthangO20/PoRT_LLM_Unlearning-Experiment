\section{Đặt vấn đề}

\begin{frame}{Đặt vấn đề}
  \begin{block}{Bối cảnh}
    \begin{itemize}
      \item Các mô hình ngôn ngữ lớn ngày càng được dùng như trợ lý tri thức trong học tập, làm việc, lập trình và nghiên cứu.
      \item Để có năng lực rộng, mô hình được huấn luyện trên lượng dữ liệu rất lớn, trong đó có thể lẫn thông tin sai lệch, riêng tư, bản quyền hoặc nhạy cảm.
      \item Sau khi mô hình đã học, việc loại bỏ một phần tri thức không còn đơn giản như xóa một file khỏi cơ sở dữ liệu.
      \item Vì vậy xuất hiện nhu cầu \term{LLM unlearning}: làm cho mô hình giảm phụ thuộc vào một nhóm tri thức không mong muốn, nhưng vẫn giữ khả năng hữu ích đã học.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Định nghĩa bài toán}
  \small
  Cho mô hình gốc $M_{\theta}$, cần tạo mô hình hoặc hệ thống sau unlearning $M'$.

  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Đầu vào bài toán}
        \begin{itemize}
          \item Forget set $D_f$: tri thức/nhiệm vụ cần làm quên.
          \item Retain set $D_r$: tri thức/nhiệm vụ cần giữ.
          \item Neighbor set $D_n$: tri thức gần miền forget nhưng an toàn.
          \item Threat model: prefix, composite, paraphrase, relearning.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Đầu ra mong muốn}
        \begin{itemize}
          \item Model, adapter hoặc inference-time pipeline.
          \item Giảm leakage trên $D_f$ và $D_f^{adv}$.
          \item Giữ utility trên $D_r$ và $D_n$.
          \item Có bằng chứng robustness, không chỉ accuracy giảm.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \[
    \min_{M'} \; \mathcal{R}_{forget}(M',D_f,D_f^{adv})
    \quad \text{s.t.} \quad
    \mathcal{U}(M',D_r,D_n)\approx \mathcal{U}(M_{\theta},D_r,D_n)
  \]
\end{frame}

\begin{frame}{Thách thức hiện tại}
  \small
  \begin{block}{Các khó khăn chính}
    \begin{itemize}
      \item Tri thức cần làm quên và tri thức hữu ích thường đan xen trong cùng biểu diễn của mô hình.
      \item Không có mô hình ``đã quên lý tưởng'' để so sánh, vì train lại LLM từ đầu gần như không khả thi.
      \item Weight-level unlearning có khả năng tác động sâu hơn nhưng dễ làm mất utility trên các tác vụ bình thường.
      \item Inference-time guard dễ triển khai hơn nhưng có thể chỉ che đầu ra, chưa chắc làm mô hình thật sự quên.
      \item Router/classifier sai ngưỡng có thể gây hai lỗi trái ngược: bỏ lọt truy vấn rủi ro hoặc từ chối quá mức.
      \item Cùng một tri thức có thể được hỏi lại bằng nhiều cách khác nhau, nên phương pháp phải bền với biến thể truy vấn.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Từ WMDP sang dữ liệu unlearning}
  \scriptsize
  \begin{block}{Điểm cần phân biệt}
    WMDP gốc \textbf{không có trường} $D_f$. $D_f$ là ký hiệu trong bài toán unlearning, được xây từ các record trắc nghiệm WMDP.
  \end{block}

  \vspace{0.25em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Xây forget set $D_f$}
        Với record $i$:
        \[
          (q_i,\; C_i=\{c_A,c_B,c_C,c_D\},\; a_i)
        \]
        tạo prompt $x_i$ gồm câu hỏi và bốn lựa chọn; target cần suppress là đáp án đúng $y_i^*=c_{a_i}$ hoặc nhãn đúng A--D.
      \end{block}
      \begin{block}{Tạo $D_f^{adv}$}
        Biến đổi \textbf{prompt} $x_i$ bằng prefix, paraphrase, composite, relearning; target vẫn là $y_i^*$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Tạo belief negatives $B_f$}
        Dùng model gốc $M_{\theta}$ sinh nhiều response cho $x_i$; giữ các response high-confidence, rủi ro hoặc gần nghĩa với target đúng để chống quên giả.
      \end{block}
      \begin{alertblock}{Retain không lấy từ WMDP}
        $D_r$ sử dụng MMLU để đo utility ngoài WMDP. $D_n$ là subset QA an toàn gần miền bio/cyber/chem.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Nghiên cứu liên quan}

\begin{frame}{PoRT: Post Judgment and Multi-round Thinking}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \begin{block}{Vấn đề paper giải quyết}
        \begin{itemize}
          \item Các phương pháp pre-filtering chỉ kiểm tra đầu vào nên dễ bị prefix/composite attack.
          \item PoRT chuyển trọng tâm từ \textit{pre-filtering} sang \textit{post-judgment}.
          \item Ý tưởng chính: xét cả câu hỏi đã làm sạch và câu trả lời mô hình sinh ra.
        \end{itemize}
      \end{block}
      \begin{alertblock}{Đóng góp chính}
        Robust unlearning framework không cần sửa trọng số, gồm prompt cleaning, post-judgment và selective multi-round thinking.
      \end{alertblock}
    \end{column}
    \begin{column}{0.56\textwidth}
      \includegraphics[width=\linewidth]{papers/port_fig1_workflow.png}
      \smallsource{Chen et al., ICLR 2026, Figure 1}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{PoRT: cách hoạt động và nhận xét}
  \scriptsize
  \begin{tabularx}{\textwidth}{l Y}
    \toprule
    \textbf{Module} & \textbf{Vai trò trong phương pháp} \\
    \midrule
    IPC & Tạo prompt few-shot từ demonstration library để làm sạch truy vấn và sinh initial response. \\
    \rowrule
    Post-Judgment & Đánh giá cặp \code{(cleaned question, response)} thay vì chỉ đánh giá prompt đầu vào. \\
    \rowrule
    SMT & Với response rủi ro hoặc không chắc chắn, kích hoạt self-correction/multi-round thinking rồi tích hợp câu trả lời cuối. \\
    \bottomrule
  \end{tabularx}

  \vspace{0.45em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ưu điểm}
        \begin{itemize}
          \item Bám sát threat model thực tế: prefix và composite prompts.
          \item Dễ gắn vào mô hình đóng/API vì không cần cập nhật trọng số.
          \item Post-judgment giảm rủi ro bỏ sót khi prompt dài hoặc bị nhiễu.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Hạn chế}
        \begin{itemize}
          \item Là inference-time defense, chưa xóa tri thức khỏi base model.
          \item Phụ thuộc chất lượng judge, demonstration library và confidence threshold.
          \item Có thêm latency do post-judgment và rethink.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LLM Beliefs: ý tưởng tổng quan}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \begin{block}{Vấn đề paper giải quyết}
        \begin{itemize}
          \item Suppress target response đơn lẻ có thể chỉ tạo cảm giác đã quên.
          \item Xác suất bị dồn sang các câu trả lời khác nhưng gần nghĩa với target.
          \item Paper gọi hiện tượng này là \term{squeezing effect}.
        \end{itemize}
      \end{block}
      \begin{alertblock}{Đóng góp chính}
        Dùng \term{model beliefs}: các high-confidence generations của chính mô hình để chống spurious unlearning.
      \end{alertblock}
    \end{column}
    \begin{column}{0.56\textwidth}
      \includegraphics[width=\linewidth]{papers/beliefs_fig1_overview.png}
      \smallsource{Li et al., ICLR 2026, Figure 1}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LLM Beliefs: cách hoạt động và nhận xét}
  \scriptsize
  \begin{tabularx}{\textwidth}{l Y}
    \toprule
    \textbf{Thành phần} & \textbf{Vai trò trong phương pháp} \\
    \midrule
    Model beliefs & Các token/sequence mà mô hình gốc sinh ra với xác suất cao quanh vùng target cần quên. \\
    \rowrule
    BS-T & Bootstrapping ở token level: suppress các token high-confidence liên quan đến target. \\
    \rowrule
    BS-S & Bootstrapping ở sequence level: thêm các generations high-confidence vào dữ liệu unlearning. \\
    \bottomrule
  \end{tabularx}

  \vspace{0.45em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ưu điểm}
        \begin{itemize}
          \item Chỉ ra cơ chế quên giả ở mức phân phối xác suất.
          \item Giảm leakage qua paraphrase/semantic alternatives.
          \item Bổ sung góc nhìn nội tại hơn so với chỉ chặn đầu ra.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Hạn chế}
        \begin{itemize}
          \item Cần sinh belief set đủ bao phủ, làm tăng chi phí.
          \item Nếu belief set thiếu biến thể quan trọng, leakage vẫn còn.
          \item Suppress quá rộng có thể ảnh hưởng tri thức hợp pháp gần miền.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Khoảng trống nghiên cứu rút ra}
  \small
  \begin{itemize}
    \item \term{PoRT} xử lý tốt robust inference-time defense, nhưng chưa can thiệp vào tri thức bên trong base model.
    
    \item \term{LLM Beliefs} giảm hiện tượng quên giả bằng belief negatives, nhưng chưa phải một pipeline phòng thủ hoàn chỉnh ở lúc inference.
    
    \item Cả hai hướng đều phụ thuộc vào coverage: demonstration library, judge, belief generations hoặc dữ liệu biến thể.
    
    \item Utility gần miền cần quên vẫn là điểm khó: suppress quá mạnh có thể ảnh hưởng tri thức hợp pháp lân cận.
  \end{itemize}

  \vspace{0.4em}
  \begin{alertblock}{Ý tưởng giải pháp}
    Kết hợp unlearning nội tại bằng adapter với post-judgment kiểu PoRT; bổ sung belief negatives và retain objective cho dữ liệu lân cận an toàn.
  \end{alertblock}
\end{frame}

\begin{frame}{Kiến trúc SAFE-PoRT chi tiết}
  \scriptsize
  \begin{tabularx}{\textwidth}{l Y Y}
    \toprule
    \textbf{Khối} & \textbf{Thành phần} & \textbf{Vai trò} \\
    \midrule
    Data construction & $D_f$, $D_f^{adv}$, $D_r$, $D_n$, $B_f$ & Tạo forget/retain/neighbor set và mở rộng bằng prefix, paraphrase, composite, relearning probes \\
    \rowrule
    Adapter unlearning & LoRA $\Delta$, NPO loss, KL retain/neighbor, representation loss, smoothness & Giảm khả năng truy hồi tri thức nguy hiểm nhưng hạn chế làm hỏng utility \\
    \rowrule
    Belief negatives & High-confidence generations từ $M_\theta$ & Giảm squeezing effect: không chỉ suppress đáp án gốc mà cả biến thể cùng nghĩa \\
    \rowrule
    Post-judgment guard & Risk score, entropy, confidence, disagreement & Kiểm tra cặp \code{(prompt, answer)} trước khi trả kết quả \\
    \rowrule
    Selective routing & Safe route hoặc rethink/refusal & Giảm leakage còn sót sau adapter, tránh always-refuse bằng threshold calibration \\
    \bottomrule
  \end{tabularx}

  \vspace{0.45em}
  \begin{block}{Luồng huấn luyện và suy luận}
    Offline: tạo $D_f^{adv}$ và $B_f$ $\rightarrow$ train LoRA adapter $\Delta$ $\rightarrow$ calibrate guard. 
    Inference: $M'=M_\theta+\Delta$ sinh candidate answer $\rightarrow$ post-judge $\rightarrow$ trả lời an toàn hoặc rethink/refuse.
  \end{block}
\end{frame}

\begin{frame}{Hàm mục tiêu đề xuất}
  \small
  Train adapter $\Delta$ để thu được $M' = M_{\theta} + \Delta$.

  \vspace{0.4em}
  \[
  \mathcal{L} =
  \lambda_1 \mathcal{L}_{NPO}(D_f \cup D_f^{adv} \cup B_f)
  + \lambda_2 KL(M' || M_{\theta}; D_r)
  + \lambda_3 KL(M' || M_{\theta}; D_{neighbor})
  + \lambda_4 \mathcal{L}_{rep}
  + \lambda_5 \mathcal{L}_{smooth}
  \]

  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Thành phần quên}
        $D_f$ được xây từ WMDP bằng cách xem đáp án đúng của mỗi câu hỏi trắc nghiệm là target cần suppress; $B_f$ là response high-confidence do model gốc sinh ra.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Thành phần giữ}
        $D_r$ giữ năng lực chung; $D_{neighbor}$ giữ kiến thức an toàn gần miền nguy hiểm.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Post-judgment router}
  \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Quy trình}
        \begin{enumerate}
          \item Chuẩn hóa/clean prompt.
          \item Sinh initial answer bằng $M'$.
          \item Judge cặp \code{(prompt, answer)}.
          \item Nếu rủi ro hoặc confidence thấp: rethink/refuse.
          \item Nếu an toàn: trả lời trực tiếp.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Tín hiệu router}
        Risk score của prompt/response, entropy đáp án A-D, confidence, disagreement giữa base model và adapter, lịch sử rethink.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

    \item Đã khảo sát các hướng liên quan, đặc biệt PoRT và LLM Beliefs, để rút ra khoảng trống: output suppression chưa đủ, MCQ-only evaluation hẹp, router calibration khó.

    \item Đã hoàn thiện khung thực nghiệm gồm ablation, so sánh với hai SOTA, visualize, phân tích trade-off và failure cases.
